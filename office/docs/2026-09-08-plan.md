**Goal:** 픽셀 아트 가상 사무실에서 AI 동료 4명과 회의하고, 회의 결과가 업무 보드 카드로 쌓이는 로컬 웹앱을 만든다. 기존 `rag_search` 코드는 건드리지 않는다.

```python
import json
import re
from typing import Any

from config import settings

_NAME_RE = re.compile(r"^[a-z_]+\.json$")


def _target(name: str):
    if not _NAME_RE.match(name):
        raise ValueError(f"bad data file name: {name!r}")
    return settings.data_dir / name


def read_json(name: str, default: Any) -> Any:
    p = _target(name)
    if not p.exists():
        return default
    try:
        return json.loads(p.read_text(encoding="utf-8"))
    except (json.JSONDecodeError, OSError):
        return default


def write_json(name: str, obj: Any) -> None:
    p = _target(name)
    tmp = p.with_suffix(".json.tmp")
    tmp.write_text(json.dumps(obj, ensure_ascii=False, indent=2), encoding="utf-8")
    tmp.replace(p)
```

```python
import subprocess
from pathlib import Path
from config import settings

_SKIP = {".git", "venv", "node_modules", "__pycache__", "office", "target", "build", ".idea", ".vscode"}


def _git_log() -> str:
    try:
        out = subprocess.run(
            ["git", "log", "--oneline", "-20"],
            cwd=settings.repo_root_path, capture_output=True, text=True, timeout=5,
        )
        return out.stdout.strip() or "(git log 없음)"
    except (OSError, subprocess.SubprocessError):
        return "(git log 조회 실패)"


def _tree(root: Path, depth: int = 2) -> str:
    lines = []
    def walk(d: Path, prefix: str, level: int):
        if level > depth:
            return
        for child in sorted(d.iterdir()):
            if child.name in _SKIP or child.name.startswith("."):
                continue
            lines.append(f"{prefix}{child.name}{'/' if child.is_dir() else ''}")
            if child.is_dir():
                walk(child, prefix + "  ", level + 1)
    walk(root, "", 1)
    return "\n".join(lines)


def assemble_brief() -> str:
    brief_md = ""
    p = Path(__file__).resolve().parent.parent / "context" / "brief.md"
    if p.exists():
        brief_md = p.read_text(encoding="utf-8")
    return (
        f"{brief_md}\n\n"
        f"## 최근 커밋 (git log -20)\n{_git_log()}\n\n"
        f"## 레포 구조\n{_tree(settings.repo_root_path)}"
    )
```

- [ ] **Step 1: 작성 (`llm_service.py` 의 `_stream_groq` 패턴)**

```python
import json
from typing import Iterator
from config import settings


class GroqError(Exception):
    pass


def _client():
    from groq import Groq
    if not settings.groq_api_key:
        raise GroqError("GROQ_API_KEY 미설정")
    return Groq(api_key=settings.groq_api_key)


def _kwargs(messages, **extra):
    kw = dict(model=settings.office_model, messages=messages, temperature=0.6, **extra)
    if "gpt-oss" in settings.office_model:
        kw["reasoning_effort"] = "low"
    return kw


def stream_turn(messages) -> Iterator[str]:
    try:
        stream = _client().chat.completions.create(stream=True, max_tokens=320, **_kwargs(messages))
        for chunk in stream:
            delta = chunk.choices[0].delta if chunk.choices else None
            if delta and delta.content:
                yield delta.content
    except GroqError:
        raise
    except Exception as e:
        raise GroqError(str(e)) from e


def complete_json(messages) -> dict:
    try:
        resp = _client().chat.completions.create(
```

```python
import time, uuid
from paths import read_json, write_json
from personas import ASSIGNEES

TAGS = {"test", "security", "feature", "fix", "spike"}
COLUMNS = ("todo", "doing", "review", "done")
_FILE = "board.json"


def list_cards():
    return read_json(_FILE, [])


def _save(cards):
    write_json(_FILE, cards)


def _validate(assignee, tag, column):
    if tag not in TAGS:
        raise ValueError(f"bad tag: {tag}")
    if column not in COLUMNS:
        raise ValueError(f"bad column: {column}")
    if assignee not in ASSIGNEES:
        raise ValueError(f"bad assignee: {assignee}")


def add_card(title, detail="", assignee="후니", tag="feature", draft_snippet=None, column="todo"):
    if not title or len(title) > 200:
        raise ValueError("title 1~200자")
    if tag in {"test", "security"}:
        assignee = "후니"
    _validate(assignee, tag, column)
    card = {
        "id": uuid.uuid4().hex, "title": title, "detail": detail[:2000],
        "assignee": assignee, "tag": tag, "draft_snippet": draft_snippet,
        "column": column, "created_at": time.time(),
    }
    cards = list_cards()
    cards.append(card)
    _save(cards)
    return card


def patch_card(card_id, *, column=None, assignee=None, tag=None):
    cards = list_cards()
    for c in cards:
        if c["id"] == card_id:
            if column is not None:
                if column not in COLUMNS:
                    raise ValueError(f"bad column: {column}")
                c["column"] = column
            if assignee is not None:
                if assignee not in ASSIGNEES:
                    raise ValueError(f"bad assignee: {assignee}")
                c["assignee"] = assignee
            if tag is not None:
                if tag not in TAGS:
                    raise ValueError(f"bad tag: {tag}")
                c["tag"] = tag
            _save(cards)
            return c
    raise KeyError(card_id)
```

```python
import json, re, time, uuid
from config import settings
import personas, context, groq_client, board

_MDIR = settings.data_dir / "meetings"
_MID_RE = re.compile(r"^[0-9a-f]{32}$")


def _mpath(mid):
    if not _MID_RE.match(mid):
        raise ValueError(f"bad meeting id: {mid!r}")
    _MDIR.mkdir(exist_ok=True)
    return _MDIR / f"{mid}.json"


def get_meeting(mid):
    p = _mpath(mid)
    return json.loads(p.read_text(encoding="utf-8")) if p.exists() else None


def _save(rec):
    _mpath(rec["meeting_id"]).write_text(
        json.dumps(rec, ensure_ascii=False, indent=2), encoding="utf-8")


def _retry_stream(messages, tries=2):
    last = None
    for i in range(tries):
        try:
            return "".join(groq_client.stream_turn(messages))
        except groq_client.GroqError as e:
            last = e
            time.sleep(1.0 * (i + 1))
    raise last


def _first_code_block(text):
    m = re.search(r"```(\w+)?\n(.*?)```", text, re.S)
    return {"lang": m.group(1) or "text", "code": m.group(2).strip()} if m else None


def run_meeting(topic, rounds=2):
    rounds = max(1, min(3, int(rounds)))
    topic = (topic or "").strip()[:2000]
    if not topic:
        yield {"type": "error", "message": "주제가 비었습니다"}
        return
    mid = uuid.uuid4().hex
    rec = {"meeting_id": mid, "topic": topic, "started_at": time.time(),
           "turns": [], "summary": "", "cards": [], "status": "running"}
    _save(rec)
    yield {"type": "start", "meeting_id": mid}
    brief = context.assemble_brief()
    seq = 0
    try:
        for _ in range(rounds):
            for c in personas.COLLEAGUES:
                text = _retry_stream(personas.turn_messages(c, brief, rec["turns"], topic))
                seq += 1
                turn = {"speaker": c.nick, "text": text.strip(), "seq": seq}
                rec["turns"].append(turn); _save(rec)
                yield {"type": "turn", **turn}
        data = groq_client.complete_json(
            personas.extract_messages(brief, rec["turns"], topic))
        rec["summary"] = data.get("summary", "")
        yield {"type": "summary", "text": rec["summary"]}
        made = []
        for item in data.get("action_items", []):
            try:
                snip = item.get("draft_snippet")
                if not snip:
                    for t in rec["turns"]:
                        snip = _first_code_block(t["text"])
                        if snip:
                            break
                made.append(board.add_card(
                    title=item["title"][:200], detail=item.get("detail", ""),
                    assignee=item.get("assignee", "후니"), tag=item.get("tag", "feature"),
                    draft_snippet=snip))
            except (KeyError, ValueError):
                continue
        rec["cards"] = made; rec["status"] = "done"; _save(rec)
        yield {"type": "cards", "cards": made}
        yield {"type": "done"}
    except groq_client.GroqError as e:
        rec["status"] = "incomplete"; _save(rec)
        yield {"type": "error", "message": f"Groq 오류: {e}"}
```

```python
from pydantic import BaseModel
import meeting, state

    topic: str
    rounds: int = 2


@router.post("/meeting")
def start_meeting(req: MeetingReq):
    if state.get_state().get("meeting_id"):
        raise HTTPException(409, "이미 진행 중인 회의가 있습니다")

    def gen():
        for ev in meeting.run_meeting(req.topic, req.rounds):

                state.set_meeting(ev["meeting_id"])
            yield f"data: {json.dumps(ev, ensure_ascii=False)}\n\n"
        state.dismiss_all()

    return StreamingResponse(gen(), media_type="text/event-stream")


@router.get("/meeting/{mid}")
```

```js
window.Office = (() => {
  const TILE = 32, ctx = document.getElementById("floor").getContext("2d");
  let actors = {}, bubbles = [];
  const COLORS = {Victoria:"#c94", Sophia:"#4a9", Michelle:"#a4c", Chloe:"#9c4", 후니:"#49c"};
  function render(state) {
    actors = {};
    for (const [nick, a] of Object.entries(state.actors))
      actors[nick] = {...a, tx: a.x, ty: a.y, cx: a.x, cy: a.y};
  }
  function say(nick, text) { bubbles.push({nick, text, until: performance.now() + 3000}); }
  function tick() {
    ctx.fillStyle = "#1c1f26"; ctx.fillRect(0,0,640,360);
    for (const [nick, a] of Object.entries(actors)) {
      a.cx += (a.tx - a.cx) * 0.1; a.cy += (a.ty - a.cy) * 0.1;
      const px = a.cx*TILE+16, py = a.cy*TILE+16;
      ctx.fillStyle = COLORS[nick] || "#888";
      ctx.beginPath(); ctx.arc(px, py, 12, 0, 7); ctx.fill();
      ctx.fillStyle = "#ddd"; ctx.font = "10px monospace"; ctx.textAlign = "center";
      ctx.fillText(nick, px, py+24);
      ctx.fillText(a.status, px, py-18);
    }
    const now = performance.now();
    bubbles = bubbles.filter(b => b.until > now);
    for (const b of bubbles) {
      const a = actors[b.nick]; if (!a) continue;
      const px = a.cx*TILE+16, py = a.cy*TILE+16;
      ctx.fillStyle = "#fff"; ctx.fillRect(px+14, py-40, 180, 30);
      ctx.fillStyle = "#111"; ctx.textAlign = "left";
      ctx.fillText(b.text.slice(0, 40), px+18, py-22);
    }
    requestAnimationFrame(tick);
  }
  requestAnimationFrame(tick);
  return {render, say};
})();
```

```js
window.Board = (() => {
  const el = document.getElementById("board");
  const COLS = [["todo","할 일"],["doing","진행 중"],["review","리뷰 대기"],["done","완료"]];
  async function refresh() {
    const cards = await fetch("/board/cards").then(r => r.json());
    el.innerHTML = "";
    for (const [key, label] of COLS) {
      const col = document.createElement("div"); col.className = "col";
      col.innerHTML = `<h4>${label}</h4>`;
      col.ondragover = e => e.preventDefault();
      col.ondrop = async e => {
        const id = e.dataTransfer.getData("id");
        await fetch(`/board/cards/${id}`, {method:"PATCH", headers:{"content-type":"application/json"},
          body: JSON.stringify({column: key})});
        refresh();
      };
      for (const c of cards.filter(c => c.column === key)) {
        const card = document.createElement("div");
        card.className = "card"; card.draggable = true;
        card.ondragstart = e => e.dataTransfer.setData("id", c.id);
        card.innerHTML = `<span class="tag ${c.tag}">${c.tag}</span> ${c.title}
          <small>@${c.assignee}</small>` +
          (c.draft_snippet ? `<pre>${c.draft_snippet.code.replace(/</g,"&lt;")}</pre>` : "");
        col.appendChild(card);
      }
      el.appendChild(col);
    }
  }
  return {refresh};
})();
```

```js
document.getElementById("summon").onclick = async () => {
  const topic = document.getElementById("topic").value.trim();
  if (!topic) return;
  await fetch("/office/summon", {method:"POST"}); await poll();
  const resp = await fetch("/meeting", {method:"POST", headers:{"content-type":"application/json"},
    body: JSON.stringify({topic, rounds: 2})});
  if (resp.status === 409) { setStatus("이미 회의 중"); return; }
  const reader = resp.body.getReader(); const dec = new TextDecoder(); let buf = "";
  for (;;) {
    const {value, done} = await reader.read(); if (done) break;
    buf += dec.decode(value, {stream:true});
    let i;
    while ((i = buf.indexOf("\n\n")) >= 0) {
      const line = buf.slice(0, i).replace(/^data: /, ""); buf = buf.slice(i + 2);
      const ev = JSON.parse(line);
      if (ev.type === "turn") { Chat.add(ev.speaker, ev.text); Office.say(ev.speaker, ev.text); }
      else if (ev.type === "summary") Chat.add("📋 요약", ev.text);
      else if (ev.type === "cards") Board.refresh();
      else if (ev.type === "error") setStatus("오류: " + ev.message);
    }
  }
  await poll();
};
function setStatus(t){ document.getElementById("status").textContent = t; }
```
